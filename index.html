function reply(index){
  const d = dialogs[step];
  addMessage('rustam', d.rustam[index]);
  setTimeout(()=>{
    addMessage('dobro', d.dobro[index]);
    step++;
    if(dialogs[step]){
      renderChoices(dialogs[step].rustam);
    }else{
      endGame();
    }
  },1500);
}

function addMessage(sender, text){
  const msg = document.createElement('div');
  msg.className = 'message ' + (sender === 'dobro' ? 'from-dobro' : 'from-rustam');
  msg.innerText = text;
  if(sender==='dobro'){
    const nick = document.createElement('div');
    nick.className = 'nick';
    nick.innerText = 'Доброжелатель';
    chat.appendChild(nick);
  }
  chat.appendChild(msg);
  window.scrollTo(0,document.body.scrollHeight);
}

function renderChoices(options){
  choices.innerHTML = '';
  options.forEach((opt, i)=>{
    const btn = document.createElement('button');
    btn.innerText = opt;
    btn.onclick = ()=>reply(i);
    choices.appendChild(btn);
  });
}

function endGame(){
  choices.innerHTML = '';
  addMessage('dobro', "Скоро я вернусь. Будь готов, Рустам. Это только начало. ;)");
}
